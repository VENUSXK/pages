```sh title="shell"   tab="渲染 command"
.venv/bin/python -m gfields.render \
  results/checkpoints/[checkpoint_name]/optimized_asset_pyr1 \
  --multiple_scattering \
  --use_perspective_cam \
  --cam_fov 60 \
  --cam_distance 2.5 \
  --cam_count 8 \
  --cam_index 0 \
  --cam_elevation 45 \
  --cam_azimuth_span 360 \
  --cam_res_x 4096 \
  --cam_res_y 4096 \
  --spp 64 \
  --max_depth 10 \
  --rr_depth 10 \
  --rr_nee_depth 64 \
  --max_depth_primitive 256 \
  --rr_depth_primitive 256 \
  --max_overlaps 256 \
  --solver_type bisection \
  --solver_max_iterations 4 \
  --stochastic_selection_criteria deterministic \
  --stochastic_selection_criteria_nee deterministic \
  --opacity_scaler 1.0 \
  --albedo 1.0 \
  --envmap data/envmaps/qwantani_dusk_2_1k.exr \
  --envmap_scale 1.0 \
  --output results/renders/[render_dir]
  # --max_level 1 \ # gaussian or gaussian+gabor
```
